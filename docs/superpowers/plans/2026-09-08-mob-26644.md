`.github/workflows/build-demo-application.yml`:

```yaml
name: Build ReactNativeDemo

on:
  pull_request:
    branches:
      - master
    types:
      - synchronize

concurrency:
  group: react-native-build-demo-${{ github.ref }}-${{ github.workflow }}
  cancel-in-progress: true

permissions:
  contents: read
  id-token: write

jobs:
  build-android:
    name: Build (ReactNativeDemo, Android)
    if: ${{ startsWith(github.event.pull_request.head.ref, 'release/') }}
    runs-on: self-runner-node-high
    timeout-minutes: 30

    steps:
      - name: Create GitHub App token
        id: github-app-token
        uses: actions/create-github-app-token@v2
        with:
          owner: ${{ github.repository_owner }}
          app-id: ${{ secrets.InsiderActionsApplication_APP_ID }}
          private-key: ${{ secrets.InsiderActionsApplication_PRIVATE_KEY }}

      - name: Checkout Repository (react-native-insider)
        uses: actions/checkout@v6
        with:
          fetch-depth: 0
          token: ${{ steps.github-app-token.outputs.token }}
          path: SDKs

      - name: Checkout Repository (ReactNativeDemo)
        uses: actions/checkout@v6
        with:
          repository: useinsider/ReactNativeDemo
          ref: main
          token: ${{ steps.github-app-token.outputs.token }}
          path: ReactNativeDemo

      - name: Print Build Context
        shell: bash
        run: |
          set -Eeuo pipefail
          RELEASE_VERSION="${GITHUB_HEAD_REF#release/}"
          DEMO_SHA="$(git -C "${GITHUB_WORKSPACE}/ReactNativeDemo" rev-parse HEAD)"
          echo "::notice title=Build Context::release ${RELEASE_VERSION} against ReactNativeDemo ${DEMO_SHA}"
          echo "ℹ️ Release version: ${RELEASE_VERSION}"
          echo "ℹ️ ReactNativeDemo commit: ${DEMO_SHA}"

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'

      - name: Setup Node
        uses: actions/setup-node@v5
        with:
          token: ${{ steps.github-app-token.outputs.token }}
          node-version: 24
          registry-url: https://registry.npmjs.org
          package-manager-cache: false

      - name: Setup Android SDK
        uses: android-actions/setup-android@v3
        with:
          accept-android-sdk-licenses: true
          log-accepted-android-sdk-licenses: false
          cmdline-tools-version: 13114758
          packages: 'platform-tools platforms;android-36 build-tools;36.0.0'

      - name: Setup Gradle
        uses: gradle/actions/setup-gradle@v3
        with:
          cache-disabled: false
          cache-read-only: false

      - name: Link SDK Into Demo
        working-directory: ${{ github.workspace }}/ReactNativeDemo
        shell: bash
        run: |
          set -Eeuo pipefail
          TARBALL="$(npm pack "${GITHUB_WORKSPACE}/SDKs" | tail -n1)"
          echo "ℹ️ Packed SDK: ${TARBALL}"
          npm install "${TARBALL}" --no-audit --no-fund
          node -e "const p=require('./package.json');console.log('ℹ️ Linked dependency:', p.dependencies['react-native-insider'])"

      - name: Build Application (Android)
        working-directory: ${{ github.workspace }}/ReactNativeDemo/android
        shell: bash
        run: |
          set -Eeuo pipefail
          chmod +x ./gradlew
          ./gradlew assembleDebug --no-daemon --stacktrace
```

`publish-release.yml` içinde `publish-release-nh` job'ının **önüne**:

```yaml
  update-demo-application:
    name: Update Demo Application (ReactNativeDemo)
    if: ${{ needs.publish-release.outputs.state == 'success' }}
    needs: [create-deployment, publish-release]
    runs-on: self-runner-node-high
    continue-on-error: true
    timeout-minutes: 20

    steps:
      - name: Create GitHub App token
        id: github-app-token
        uses: actions/create-github-app-token@v2
        with:
          owner: ${{ github.repository_owner }}
          app-id: ${{ secrets.InsiderActionsApplication_APP_ID }}
          private-key: ${{ secrets.InsiderActionsApplication_PRIVATE_KEY }}

      - name: Checkout Repository (react-native-insider)
        uses: actions/checkout@v4
        with:
          fetch-depth: 1
          token: ${{ steps.github-app-token.outputs.token }}
          path: SDKs

      - name: Checkout Repository (ReactNativeDemo)
        uses: actions/checkout@v4
        with:
          repository: useinsider/ReactNativeDemo
          ref: main
          token: ${{ steps.github-app-token.outputs.token }}
          path: ReactNativeDemo

      - name: Configure Git Properties
        working-directory: ${{ github.workspace }}/ReactNativeDemo
        run: |
          set -Eeuo pipefail
          git config user.name "insider-automation"
          git config user.email "devops@useinsider.com"
          git config --add safe.directory "${GITHUB_WORKSPACE}/ReactNativeDemo"
          git config gpg.format openpgp
          git config commit.gpgsign true

      - name: Import GPG Key
        uses: crazy-max/ghaction-import-gpg@v6
        with:
          gpg_private_key: ${{ secrets.INSIDER_AUTOMATION_GITHUB_GPG_PRIV_KEY }}
          passphrase: ${{ secrets.INSIDER_AUTOMATION_GITHUB_GPG_PASSPHRASE }}
          git_user_signingkey: true
          git_commit_gpgsign: true
          trust_level: 5
          workdir: ${{ github.workspace }}/ReactNativeDemo

      - name: Setup Node
        uses: actions/setup-node@v5
        with:
          token: ${{ steps.github-app-token.outputs.token }}
          node-version: 24
          registry-url: https://registry.npmjs.org
          package-manager-cache: false

      - name: Bump Dependency
        id: bump
        working-directory: ${{ github.workspace }}/ReactNativeDemo
        shell: bash
        env:
          VERSION: ${{ needs.create-deployment.outputs.version }}
        run: |
          set -Eeuo pipefail
          CURRENT="$(node -p "require('./package.json').dependencies['react-native-insider']")"
          echo "ℹ️ Current: ${CURRENT}, target: ${VERSION}"

          attempt=1
          until npm install "react-native-insider@${VERSION}" --save-exact --no-audit --no-fund; do
            if [[ "${attempt}" -ge 3 ]]; then
              echo "⛔ Error: react-native-insider@${VERSION} npm registry'de 3 denemede bulunamadı."
              exit 1
            fi
            echo "⚠️ Warning: deneme ${attempt} başarısız, registry propagation için 15 sn bekleniyor..."
            attempt=$(( attempt + 1 ))
            sleep 15
          done

          if git diff --quiet -- package.json package-lock.json; then
            echo "ℹ️ ReactNativeDemo zaten ${VERSION} pinliyor, commit atlanıyor."
            echo "committed=false" >> "$GITHUB_OUTPUT"
            exit 0
          fi

          git add package.json package-lock.json
          git commit -m "chore(deps): bump react-native-insider to ${VERSION}"
          git push origin HEAD:main
          echo "committed=true" >> "$GITHUB_OUTPUT"
          echo "✅ ReactNativeDemo main ${VERSION} sürümüne güncellendi."

      - name: Notify Slack (on failure)
        if: failure()
        working-directory: ${{ github.workspace }}/SDKs
        shell: bash
        env:
          VERSION: ${{ needs.create-deployment.outputs.version }}
        run: |
          set -Eeuo pipefail
          test -x devops/slack_notifier.sh || chmod +x devops/slack_notifier.sh
          bash devops/slack_notifier.sh \
            --message "ReactNativeDemo bump failed for react-native-insider ${VERSION}. Run: ${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}"
```

import UserNotifications
import InsiderMobileAdvancedNotification

// FIXME-INSIDER: Please change with your app group.
private let appGroup = "group.com.useinsider.reactnativedemo"

final class NotificationService: UNNotificationServiceExtension {

    private var contentHandler: ((UNNotificationContent) -> Void)?
    private var bestAttemptContent: UNMutableNotificationContent?

    override func didReceive(
        _ request: UNNotificationRequest,
        withContentHandler contentHandler: @escaping (UNNotificationContent) -> Void
    ) {
        self.contentHandler = contentHandler

        guard let content = request.content.mutableCopy() as? UNMutableNotificationContent else {
            contentHandler(request.content)
            return
        }
        bestAttemptContent = content

        // MARK: You can customize these.
        let nextButtonText = ">>"
        let goToAppText = "Launch App"

        InsiderPushNotification.showInsiderRichPush(
            content,
            appGroup: appGroup,
            nextButtonText: nextButtonText,
            goToAppText: goToAppText
        ) { attachment in
            if let attachment {
                content.attachments = [attachment]
            }
            contentHandler(content)
        }
    }

    override func serviceExtensionTimeWillExpire() {
        if let contentHandler, let bestAttemptContent {
            contentHandler(bestAttemptContent)
        }
    }
}
